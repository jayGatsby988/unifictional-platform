/**
 * Device/connection checks for the site's heaviest optional extras.
 *
 * Live embeds load entire third-party websites, so they're reserved for
 * machines and connections that can take it: a real desktop viewport, a decent
 * CPU/RAM, a fast connection and no data-saver. Everyone else sees the
 * (identical-looking) screenshots. Browsers that don't expose a signal (Safari,
 * Firefox) are judged on the signals they do expose.
 */
type Conn = { saveData?: boolean; effectiveType?: string; downlink?: number };
type Nav = Navigator & { connection?: Conn; deviceMemory?: number };

export function canRunLiveEmbeds(): boolean {
  if (typeof window === "undefined") return false;
  if (!window.matchMedia("(min-width: 1024px)").matches) return false;
  if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) return false;
  const nav = navigator as Nav;
  const c = nav.connection;
  if (c?.saveData) return false;
  if (c?.effectiveType && c.effectiveType !== "4g") return false;
  if (typeof c?.downlink === "number" && c.downlink > 0 && c.downlink < 5) return false;
  if (typeof nav.deviceMemory === "number" && nav.deviceMemory < 8) return false;
  if (typeof navigator.hardwareConcurrency === "number" && navigator.hardwareConcurrency < 8)
    return false;
  return true;
}

/** True on data-saver / slow connections or low-memory devices. (Core count
 *  isn't used here: iOS Safari under-reports it, which would wrongly
 *  downgrade every iPhone.) */
export function isConstrained(): boolean {
  if (typeof window === "undefined") return false;
  const nav = navigator as Nav;
  const c = nav.connection;
  return (
    !!c?.saveData ||
    (c?.effectiveType != null && c.effectiveType !== "4g") ||
    (typeof c?.downlink === "number" && c.downlink > 0 && c.downlink < 1.5) ||
    (typeof nav.deviceMemory === "number" && nav.deviceMemory <= 2)
  );
}

/** Run `fn` once the page has fully loaded and the main thread is idle. */
export function afterLoadIdle(fn: () => void, timeout = 2500): () => void {
  let cancelled = false;
  let idle: number | undefined;
  let t: ReturnType<typeof setTimeout> | undefined;
  const w = window as Window & {
    requestIdleCallback?: (cb: () => void, o?: { timeout: number }) => number;
    cancelIdleCallback?: (id: number) => void;
  };
  const run = () => !cancelled && fn();
  const arm = () => {
    if (cancelled) return;
    if (w.requestIdleCallback) idle = w.requestIdleCallback(run, { timeout });
    else t = setTimeout(run, 800);
  };
  if (document.readyState === "complete") arm();
  else window.addEventListener("load", arm, { once: true });
  return () => {
    cancelled = true;
    window.removeEventListener("load", arm);
    if (idle != null) w.cancelIdleCallback?.(idle);
    if (t) clearTimeout(t);
  };
}

// Generates lightweight image variants at build time (runs before `next build`).
// Originals stay the single source of truth; nothing generated needs committing.
//
//  • Portrait hero frames — phones only ever see the middle ~35% of the 16:9
//    footage, so they get a centre crop (same framing, same sharpness) at about
//    a third of the bytes and a third of the decoded memory per frame.
//  • Work-carousel thumbnails — 360px copies instead of full 1280px screenshots.
//  • Process backgrounds — WebP versions of the two JPEGs.
//
// Idempotent: existing outputs are skipped. Fails the build loudly if an output
// can't be produced, so a deploy can never ship pages pointing at missing files.
import { existsSync, mkdirSync, readdirSync } from "node:fs";
import { join } from "node:path";
import sharp from "sharp";

const PUB = "public";
const jobs = [];

// 1) Portrait hero frames: every 4th frame (the phone scrub set) + the last one.
const FRAMES = join(PUB, "frames/unifictional-v3");
const PORTRAIT = join(PUB, "frames/unifictional-v3-portrait");
const CROP_W = 560; // covers any viewport narrower than 5:8 (all phones, portrait)
mkdirSync(PORTRAIT, { recursive: true });
for (let i = 0; i <= 240; i += 4) {
  const name = `frame-${String(i).padStart(3, "0")}.webp`;
  const out = join(PORTRAIT, name);
  if (existsSync(out)) continue;
  jobs.push(async () => {
    const img = sharp(join(FRAMES, name));
    const { width, height } = await img.metadata();
    const w = Math.min(CROP_W, width);
    await img
      .extract({ left: Math.round((width - w) / 2), top: 0, width: w, height })
      .webp({ quality: 74, effort: 5 })
      .toFile(out);
  });
}

// 2) Carousel thumbnails.
const SHOW = join(PUB, "showcase");
const THUMBS = join(SHOW, "thumbs");
mkdirSync(THUMBS, { recursive: true });
for (const f of readdirSync(SHOW).filter((f) => f.endsWith(".webp"))) {
  const out = join(THUMBS, f);
  if (existsSync(out)) continue;
  jobs.push(() =>
    sharp(join(SHOW, f)).resize({ width: 360 }).webp({ quality: 72 }).toFile(out)
  );
}

for (const n of ["bg-1", "bg-2"]) {
